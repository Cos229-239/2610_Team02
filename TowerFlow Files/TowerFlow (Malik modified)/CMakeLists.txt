cmake_minimum_required(VERSION 3.16)
project(TowerFlow LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Legacy files are included by main. Do not compile individually.
add_executable(TowerFlow "TowerFlow (Malik version).cpp")
if(WIN32)
    target_sources(TowerFlow PRIVATE TowerFlowGUI.cpp)
    target_link_libraries(TowerFlow PRIVATE comctl32 user32 gdi32)
endif()
add_executable(PartTests tests/CurrentPartTests.cpp)
enable_testing()
add_test(NAME PartTests COMMAND PartTests)

add_executable(OperationsTests tests/OperationsTests.cpp)
add_test(NAME OperationsTests COMMAND OperationsTests)
